{{/* Every seminar, filterable. There is no search index and no search library. All 322 talks are rendered into the page and the box filters the rows that are already there. At this size that is faster than fetching and querying an index, it cannot drift out of sync with the data the way a generated index can, and it needs no dependency -- which matters for a site that has to survive years of nobody maintaining it. It also degrades honestly: with JavaScript off the box is hidden and you get the full list, where the browser's own find-in-page still works. Each row carries a data-s attribute with everything worth matching on, so the filter never has to walk the DOM for text. */}} {{- $seasons := dict -}} {{- range hugo.Data.seasons }}{{ $seasons = merge $seasons (dict .slug .label) }}{{ end -}} {{- $all := slice -}} {{- range $name, $rec := hugo.Data.seminars -}} {{- $all = $all | append (merge $rec (dict "id" $name)) -}} {{- end -}} {{- $all = sort $all "seminar_date" "desc" -}}
