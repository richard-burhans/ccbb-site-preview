{{/* One seminar row. Context: dict of "s" (the record), "next" (seminar_date of the next upcoming talk, for highlighting) and "now". Date rules carried over from the original seminar-tr.html: - midnight means the time is not known yet -> "[time TBA]" - 3:00 pm is the default slot, so it is left implicit; any other time is shown `status` replaces the five ad-hoc conventions the data used to carry (a reschedule field, "No seminar" as a presenter, a "CANCELLED" prefix on presenters/affiliations/title, and "[Thanksgiving break]" as a title). Status is never signalled by colour alone -- each one carries a text badge. */}} {{- $s := .s -}} {{- $season := .season_label -}} {{- $status := or $s.status "scheduled" -}} {{/* Everything worth matching on, flattened into one attribute so the search filter never has to walk the DOM for text. Dates are included in several forms because people search "2019", "March 2019" and "2019-03" alike. */}} {{- $d := time.AsTime $s.seminar_date -}} {{- $searchParts := slice (or $s.presenters "") (or $s.affiliations "") (or $s.title "") (or $s.host "") (or $s.location "") (or $s.note "") (or $season "") ($d.Format "2006") ($d.Format "January 2 2006") ($d.Format "2006-01-02") (or $status "") -}} {{- $search := delimit $searchParts " " -}} {{- $date := time.AsTime $s.seminar_date -}} {{- $isPast := lt $date .now -}} {{- $isNext := and (ne .next "") (eq $s.seminar_date .next) -}} {{- $tba := and (eq $date.Hour 0) (eq $date.Minute 0) -}} {{- $offSlot := or (ne $date.Hour 15) (ne $date.Minute 0) -}} {{- $off := eq $status "no_seminar" -}} {{- if eq $status "rescheduled" -}} To be rescheduled {{- else -}} {{ $date.Format "January 2" }} {{- if and (not $off) $tba -}} [time TBA] {{- else if and (not $off) $offSlot -}} {{ $date.Format "3:04 pm" }} {{- end -}} {{- end -}} {{- with $s.location }}{{ . }}{{ end -}} {{/* Shown only where seasons are mixed together, i.e. the search page. */}} {{- with $season }}{{ . }}{{ end -}} {{- if $isNext }} Next{{ end -}} {{- if eq $status "cancelled" }} Cancelled{{ end -}} {{- if $off -}} No seminar {{- else -}} {{- with $s.presenters -}} {{ . }} {{- else -}} {{/* A held date with no speaker confirmed yet. Saying so is better than an apparently empty row, which reads as missing data rather than as a slot deliberately reserved. */}} Speaker to be announced {{- end -}} {{- with $s.affiliations }}{{ . }}{{ end -}} {{- with $s.host }}Host: {{ . }}{{ end -}} {{- end -}} {{- with $s.title }}{{ . }}{{ end -}} {{- with $s.note }}{{ . }}{{ end -}} {{- $zoom := and (ne $status "rescheduled") (ne $status "no_seminar") $s.zoom_url -}} {{- if or $zoom $s.slides }} {{- with $s.slides }}Slides{{ end -}} {{- if $zoom }}Zoom{{ end -}} {{- end -}} {{- if $s.id }}{{ partial "edit-link.html" $s }}{{ end -}}
