{{/* One seminar row. Context: dict of "s" (the record), "next" (seminar_date of the next upcoming talk, for highlighting) and "now". Rendering rules carried over from the old seminar-tr.html: - midnight means the time is not known yet -> "[time TBA]" - 3:00 pm is the default slot, so it is left implicit; any other time is shown - a `reschedule` record shows no date at all Status is still inferred from the old ad-hoc conventions here. Phase 2 replaces that with an explicit `status` field. */}} {{- $s := .s -}} {{- $date := time.AsTime $s.seminar_date -}} {{- $isPast := lt $date .now -}} {{- $isNext := and (ne .next "") (eq $s.seminar_date .next) -}} {{- $tba := and (eq $date.Hour 0) (eq $date.Minute 0) -}} {{- $offSlot := or (ne $date.Hour 15) (ne $date.Minute 0) -}} {{/* Legacy conventions: "CANCELLED"/"CANCELED" prefixed onto presenters, and "No seminar" as a presenter name. Matched case-insensitively. */}} {{- $presenters := or $s.presenters "" -}} {{- $upper := upper $presenters -}} {{- $cancelled := or (hasPrefix $upper "CANCELLED") (hasPrefix $upper "CANCELED") -}} {{- $off := or (eq $upper "NO SEMINAR") (hasPrefix $upper "[NO SEMINAR") -}} {{- if $s.reschedule -}} To be rescheduled {{- else -}} {{ $date.Format "January 2" }} {{- if $tba -}} [time TBA] {{- else if $offSlot -}} {{ $date.Format "3:04 pm" }} {{- end -}} {{- end -}} {{- with $s.location }}{{ . }}{{ end -}} {{- if $isNext }} Next{{ end -}} {{- if $cancelled }} Cancelled{{ end -}} {{- with $presenters }}{{ . }}{{ end -}} {{- with $s.affiliations }}{{ . }}{{ end -}} {{- with $s.host }}Host: {{ . }}{{ end -}} {{- with $s.title }}{{ . }}{{ end -}} {{- $zoom := and (not $s.reschedule) $s.zoom_url -}} {{- if or $zoom $s.slides }} {{- with $s.slides }}Slides{{ end -}} {{- if $zoom }}Zoom{{ end -}} {{- end -}}
